cmake_minimum_required(VERSION 3.20)
project(personal_assistant VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_executable(assistant src/main.cpp)
target_include_directories(assistant PRIVATE vendor)
target_link_libraries(assistant PRIVATE SQLite::SQLite3 OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
  target_link_libraries(assistant PRIVATE ws2_32 crypt32)
endif()
